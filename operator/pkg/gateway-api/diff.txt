+	var grpcRouteList gatewayv1.GRPCRouteList

+		return nil
+	}
+

+			isAllowed(ctx, r.Client, gw, &route, r.logger) &&

+			filtered = append(filtered, route)
+		}
+	}

    gateway-api: align SupportedKinds handling with Gateway API spec

    to better align with the Gateway API spec.
    
    - For HTTP/HTTPS protocols, both HTTPRoute and GRPCRoute are returned by default.

    This improves conformance and prevents unexpected behavior when GRPCRoute is not installed.
    
    Fixes: #39021
